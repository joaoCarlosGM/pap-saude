import assert from "node:assert/strict";
import { after, test } from "node:test";
import { execFileSync } from "node:child_process";
import { db } from "../../src/server/db/client";
import { bootstrapSystemIam } from "../../src/server/iam/bootstrap.service";
import { actor, request } from "../../tests/mission04-b/fixtures";
import { GET } from "../../src/app/api/admin/organizations/route";
after(() => db.$disconnect());
test("PAP_MANAGER organization.read must not disclose private settings", async () => {
  assert.equal(new URL(process.env.DATABASE_URL!).pathname, "/pap_m04_b_final_review");
  assert.equal(execFileSync("docker", ["inspect", "--format", '{{ index .Config.Labels "pap.mission" }}', "pap_saude_db"], {encoding:"utf8"}).trim(), "m04-b-final-audit");
  await bootstrapSystemIam();
  const manager = await actor("PAP_MANAGER");
  await db.organization.update({where:{id:manager.unit.id}, data:{settings:{privateAuditCanary:"synthetic-only"}}});
  const response = await GET(request("/api/admin/organizations", manager.token));
  assert.equal(response.status,200);
  const body = await response.json();
  const organization = body.organizations.find((row: {id:string}) => row.id === manager.unit.id);
  assert.ok(organization);
  console.log(JSON.stringify({httpStatus:response.status,settingsReturned:Object.hasOwn(organization,"settings"),privateCanaryVisible:organization.settings?.privateAuditCanary === "synthetic-only",clinicalCountsReturned:Object.hasOwn(organization._count,"patientLinks") || Object.hasOwn(organization._count,"encounters"),returnedScalarFields:Object.keys(organization).filter(k=>!k.startsWith("_")&& k!=="parent").sort()}));
  assert.equal(Object.hasOwn(organization,"settings"),false,"Private organizational settings must be absent for PAP_MANAGER");
});
test("PAP_MANAGER cannot use organization.read for undelegated global endpoints", async () => {
  assert.equal(new URL(process.env.DATABASE_URL!).pathname, "/pap_m04_b_final_review");
  const manager = await actor("PAP_MANAGER");
  for (const path of ["audit","users","beta/users","security","beta/metrics","operations/metrics","operations/monitoring","operations/integrations"]) {
    const handler = await import(`../../src/app/api/admin/${path}/route`);
    const response = await handler.GET(request(`/api/admin/${path}`,manager.token));
    console.log(JSON.stringify({endpoint:`/api/admin/${path}`,httpStatus:response.status}));
    if (path === "users") {
      assert.equal(response.status,200);
      const body = await response.json();
      console.log(JSON.stringify({endpoint:"/api/admin/users",emptyResult:Array.isArray(body.users) && body.users.length === 0}));
      assert.deepEqual(body.users,[]);
    } else assert.equal(response.status,403,path);
  }
});
